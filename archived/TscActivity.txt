    public String SendPrinterFW(Context ctx, Uri uri) {
        if (this.OutStream == null || this.InStream == null) {
            if (this.OutStream == null) {
                Log.e("THINBTCLIENT", "OutputStream is null\r\n");
                return "-1";
            }

            if (this.InStream == null) {
                Log.e("THINBTCLIENT", "InputStream is null\r\n");
                return "-1";
            }
        }

        try {
            InputStream fis = ctx.getContentResolver().openInputStream(uri);
            byte[] data = new byte[fis.available()];
            int[] FF = new int[data.length];
            byte[] DisableESC = new byte[]{27, 33, 68};

            while(fis.read(data) != -1) {
            }

            this.OutStream.write(DisableESC);
            this.OutStream.write(data);
            this.OutStream.flush();
            fis.close();
        } catch (Exception e) {
            Log.e("THINBTCLIENT", "Exception during write\r\n");

    public String printerSN() {
        if (this.OutStream == null || this.InStream == null) {
            if (this.OutStream == null) {
                Log.e("THINBTCLIENT", "OutputStream is null\r\n");
                return "-1";
            }

            if (this.InStream == null) {
                Log.e("THINBTCLIENT", "InputStream is null\r\n");
                return "-1";
            }
        }

        try {
            String cmd = "OUT GETSETTING$(\"SYSTEM\",\"INFORMATION\",\"SERIAL\")\r\n";
            byte[] cmd_byte = cmd.getBytes("UTF-8");
            this.OutStream.write(cmd_byte);
        } catch (IOException e) {
            Log.e("THINBTCLIENT", "Exception during write\r\n");
            e.printStackTrace();
            return "-1";
        }

        try {
            Thread.sleep(300L);
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        readBuf = new byte[1024];

        try {
            while(this.InStream.available() > 0) {
                int e = this.InStream.read(readBuf);
            }
        } catch (Exception e) {
            Log.e("THINBTCLIENT", "Exception during read\r\n");
            e.printStackTrace();
            return "-1";
        }

        String SN = new String(readBuf, StandardCharsets.US_ASCII);
        return SN;
    }
